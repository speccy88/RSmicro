set(RSM_SOURCES src/rsm_runtime.c)
add_library(rsmcore STATIC ${RSM_SOURCES})
target_include_directories(rsmcore PUBLIC include)
target_compile_features(rsmcore PUBLIC c_std_99)
if(RSM_BUILD_SHARED)
 add_library(rsmcore_shared SHARED ${RSM_SOURCES})
 target_include_directories(rsmcore_shared PUBLIC include)
 set_target_properties(rsmcore_shared PROPERTIES OUTPUT_NAME rsmcore)
endif()
set(RSM_TARGETS rsmcore)
if(RSM_BUILD_SHARED)
 list(APPEND RSM_TARGETS rsmcore_shared)
endif()
if(RSM_ENABLE_STRICT_WARNINGS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 foreach(t IN LISTS RSM_TARGETS)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion -Wstrict-prototypes -Wmissing-prototypes)
 endforeach()
endif()
if(RSM_ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 foreach(t IN LISTS RSM_TARGETS)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PRIVATE -fsanitize=address,undefined)
 endforeach()
endif()
if(RSM_BUILD_TESTS)
 enable_testing()
 add_executable(rsmcore_tests tests/test_main.c)
 target_link_libraries(rsmcore_tests PRIVATE rsmcore)
 add_test(NAME rsmcore_tests COMMAND rsmcore_tests)
 add_executable(rsmcore_conformance_tests tests/test_conformance.c)
 target_link_libraries(rsmcore_conformance_tests PRIVATE rsmcore)
 add_test(NAME rsmcore_conformance_tests COMMAND rsmcore_conformance_tests)
endif()
